{#- hero 要素の直後に配置されるタブ形式のナビゲーション #} {%- set nav_items = config.extra.header_nav | default(value=[]) %} {#- section, page にいる場合、そのnav項目のセクション自身またはセクション配下にいれば current 表示する #} {%- set current_section_chain = [] %} {%- if section is defined %} {%- set current_section_chain = section.ancestors | concat(with=section.relative_path) %} {%- elif page is defined %} {%- set current_section_chain = page.ancestors %} {%- endif %} {%- set current_taxonomy_kind = "" %} {%- if taxonomy is defined %} {%- set current_taxonomy_kind = taxonomy.name %} {%- endif %} {%- if nav_items | length > 0 %}
    {%- for item in nav_items %} {%- if item.type == "section" %} {%- set resolved = get_section(path=item.path) %} {%- set href = resolved.permalink %} {%- set label = item.title | default(value=resolved.title) %} {%- set is_current = item.path in current_section_chain %} {%- elif item.type == "taxonomy" %} {%- set resolved = get_taxonomy(kind=item.kind) %} {%- set href = resolved.permalink %} {%- set label = item.title | default(value=resolved.kind.name) %} {%- set is_current = item.kind == current_taxonomy_kind %} {%- else %} {{ throw(message="header_nav: unknown type `" ~ item.type ~ "` (must be `section` or `taxonomy`)") }} {%- endif %} {{ label }} {%- endfor %}
{%- endif %}
