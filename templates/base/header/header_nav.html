{#- hero 要素の直後に配置されるタブ形式のナビゲーション #} {%- set nav_items = config.extra.header_nav | default(value=[]) %} {#- section を直接表示しているときのみ current 表示する（配下の個別ページ閲覧中はどのタブも current にしない） #} {%- set current_section_path = "" %} {%- if section is defined %} {%- set current_section_path = section.relative_path %} {%- endif %} {%- set current_taxonomy_kind = "" %} {%- if taxonomy is defined %} {%- set current_taxonomy_kind = taxonomy.name %} {%- endif %} {%- if nav_items | length > 0 %}
    {%- for item in nav_items %} {%- if item.type == "section" %} {%- set resolved = get_section(path=item.path) %} {%- set href = resolved.permalink %} {%- set label = item.title | default(value=resolved.title) %} {%- set is_current = item.path == current_section_path %} {%- elif item.type == "taxonomy" %} {%- set resolved = get_taxonomy(kind=item.kind) %} {%- set href = resolved.permalink %} {%- set label = item.title | default(value=resolved.kind.name) %} {%- set is_current = item.kind == current_taxonomy_kind %} {%- else %} {{ throw(message="header_nav: unknown type `" ~ item.type ~ "` (must be `section` or `taxonomy`)") }} {%- endif %} {{ label }} {%- endfor %}
{%- endif %}
